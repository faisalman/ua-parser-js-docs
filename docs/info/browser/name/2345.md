## Code Example

```js
import { UAParser } from 'ua-parser-js';
import { BrowserName } from 'ua-parser-js/enums';

const parser = new UAParser();
const ua = 'Mozilla/5.0 (Windows NT 10.0; WOW64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/56.0.2924.90 Safari/537.36 2345Explorer/9.2.1.17116';

const browser = parser.setUA(ua).getBrowser();

console.log(browser);
// {name: "2345", version: "9.2.1.17116", major: "9"}

console.log(browser.is(BrowserName.2345));
// true
```
